SIWES Document Tracking System (DBMS)

Architected for the LASUED SIWES Administrative Unit to transition manual paper tracking to an automated, auditable relational database workflow.

Relational Entities

Normalized schema covering Students, Supervisors, Student-Supervisor Assignments, and Document Tracking logs.

Data Integrity

Enforces primary keys, unique constraints on Matric & Staff numbers, foreign keys with ON DELETE CASCADE, and ENUM state transitions.

Operational Impact

Eliminated lost paper logbooks and reduced document retrieval and review times by over 70%.

-- SIWES Document Tracking System Schema
-- Designed for LASUED SIWES Unit
-- Created by Benjamin Opeyemi Abraham

CREATE DATABASE IF NOT EXISTS lasued_siwes_db;
USE lasued_siwes_db;

-- 1. Table for Students
CREATE TABLE IF NOT EXISTS students (
    student_id INT AUTO_INCREMENT PRIMARY KEY,
    matric_no VARCHAR(20) UNIQUE NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    department VARCHAR(100) NOT NULL,
    phone_number VARCHAR(15),
    email VARCHAR(100) UNIQUE,
    placement_company VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Table for Supervisors
CREATE TABLE IF NOT EXISTS supervisors (
    supervisor_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    staff_no VARCHAR(20) UNIQUE NOT NULL,
    department VARCHAR(100) NOT NULL,
    is_internal BOOLEAN DEFAULT TRUE -- TRUE for LASUED faculty, FALSE for Industry supervisor
);

-- 3. Table for Student-Supervisor Assignment
CREATE TABLE IF NOT EXISTS student_supervisor (
    assignment_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT NOT NULL,
    supervisor_id INT NOT NULL,
    assignment_date DATE DEFAULT (CURRENT_DATE),
    FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE,
    FOREIGN KEY (supervisor_id) REFERENCES supervisors(supervisor_id) ON DELETE CASCADE
);

-- 4. Table for Document Tracking (Logbooks, Acceptance Letters, etc.)
CREATE TABLE IF NOT EXISTS document_tracking (
    tracking_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT NOT NULL,
    document_type ENUM('Logbook', 'Acceptance Letter', 'Sc-05 Form', 'Evaluation Form') NOT NULL,
    submission_date DATE NOT NULL,
    status ENUM('Pending Review', 'Approved', 'Rejected', 'Missing') DEFAULT 'Pending Review',
    remarks TEXT,
    FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE
);
